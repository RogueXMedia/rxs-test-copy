# Applies only to this site's verified document root.
DirectoryIndex index.html
Options -Indexes

<IfModule mod_dir.c>
DirectorySlash Off
</IfModule>

<IfModule mod_rewrite.c>
RewriteEngine On
# Preserve SSL renewal paths and the canonical domain.
RewriteCond %{REQUEST_URI} !(^|/)\.well-known/
RewriteCond %{HTTP_HOST} !^roguexservices\.com$ [NC]
RewriteRule ^(.*)$ https://roguexservices.com/$1 [R=301,L]
RewriteCond %{REQUEST_URI} !(^|/)\.well-known/
RewriteCond %{HTTPS} off
RewriteCond %{HTTP:X-Forwarded-Proto} !https [NC]
RewriteRule ^(.*)$ https://roguexservices.com/$1 [R=301,L]
RewriteCond %{THE_REQUEST} \s/+index\.html[\s?] [NC]
RewriteRule ^index\.html$ https://roguexservices.com/ [R=301,L]
# Retired owner portal: return a real 404 even if old files remain on disk.
RewriteRule ^(?:manage|portal-api|directory-data|portal)(?:/|$) - [R=404,L,NC]
# Server backups (*.bak, *.bak-YYYYMMDDx) are never public (2026-10-18m).
RewriteRule \.bak(?:-[^/]*)?$ - [R=404,L,NC]
# Listing applications retired 2026-10-16; send old links to the directory.
RewriteRule ^get-listed(?:/.*)?$ https://roguexservices.com/services [R=301,L,NC]
# Old bookmarked privacy URL now uses the actual email-application policy.
RewriteRule ^privacy-policy(?:\.html)?/?$ /privacy [R=301,L]
# Serve generated content at clean URLs without a blanket SPA fallback.
RewriteCond %{REQUEST_FILENAME} -d
RewriteCond %{REQUEST_FILENAME}/index.html -f
RewriteRule ^(.+?)/?$ $1/index.html [L]
</IfModule>

ErrorDocument 404 /404.html

</FilesMatch>
